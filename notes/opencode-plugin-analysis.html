opencode 插件核心函数分析

从 open-island.js + vibe-island.js 中提取的关键函数 — 2026-07-02

以下函数从两个真实插件中提取，只保留了核心逻辑，去掉了边界处理。直接复制到 Node.js 环境就能跑，边改边试。

1. 通信：Unix Socket

插件与桌面 App 之间通过 Unix domain socket 通信。 只发不收（fire-and-forget）用 sendToSocket，等回复用 sendAndWaitResponse。
sendToSocket SOCKET open-island.js
功能：发一条 JSON 消息到 Unix socket，不等回复。
场景：SessionStart、UserPromptSubmit、PreToolUse 等不需要等回应的通知。
参数：json — 任意可序列化的对象。
返回：true（发成功）、false（发失败—socket 不通）。
超时：3 秒后自动销毁。 
// open-island.js — 发送后不等回复
function sendToSocket(json) {
  return new Promise(function(resolve) {
    try {
      const sock = connect({ path: SOCKET_PATH }, function() {
        // socket 连接成功，发送 JSON + 换行
        sock.end(encodeEnvelope(json));
      });
      // 3 秒超时，超时即认为失败
      sock.setTimeout(3000, function() { sock.destroy(); resolve(false); });
      sock.on("error", function() { resolve(false); });
    } catch { resolve(false); }
  });
}

// encodeEnvelope: 把 JSON 包装成一行，末尾加 \n
// 桌面 App 端按行读取 socket
function encodeEnvelope(command) {
  return JSON.stringify({ type: "command", command }) + "\n";
}
sendAndWaitResponse SOCKET open-island.js
功能：发一条 JSON 消息到 Unix socket，等回复。
场景：PermissionRequest（请求用户批准）、QuestionAsked（等待用户回答）。需要用户点"允许"或"拒绝"，不能 fire-and-forget。
返回：解析后的 JSON 对象，或 null（超时/失败）。
超时：5 分钟（300秒）——因为要等用户看完对话框。 
// open-island.js — 发送后收集回复
function sendAndWaitResponse(json, timeoutMs = 300000) {
  return new Promise(function(resolve) {
    try {
      const sock = connect({ path: SOCKET_PATH }, function() {
        sock.write(encodeEnvelope(json));  // 发送请求
      });
      let buf = "";
      sock.on("data", function(chunk) {
        buf += chunk.toString();
        // BridgeServer 先发 "hello"，处理完后再发响应
        // 所以收到 >= 2行时，第2行就是回复
        const lines = buf.split("\n").filter(Boolean);
        if (lines.length >= 2) {
          sock.destroy();
          try { resolve(JSON.parse(lines[1])); }
          catch { resolve(null); }
        }
      });
      sock.setTimeout(timeoutMs, function() { sock.destroy(); resolve(null); });
    } catch { resolve(null); }
  });
}

2. 事件映射：opencode → 桌面 App

这是插件的核心逻辑：把 opencode 的 event 对象转换成桌面 App 能理解的消息。 下面只列了最重要的 5 个事件分支。
mapEvent — session.created EVENT open-island.js
触发：用户在 opencode 中创建了一个新对话。
做的事情：记录 session 的工作目录（cwd），发 SessionStart 通知桌面 App。 
if (t === "session.created" && p.info) {
  const cwd = p.info.directory || "";
  sessionCwd.set(p.info.id, cwd);       // 记住这个 session 的工作目录
  return makePayload("SessionStart", p.info.id, cwd);
}
mapEvent — message.part.updated (text) EVENT open-island.js
触发：模型或用户在对话中发了一段文字。
做的事情：判断是谁发的（meta.role），如果是用户→发 UserPromptSubmit（含 prompt 文本），如果是助手→只记录文本但不发事件。 
if (t === "message.part.updated" && p.part?.type === "text" && p.part?.messageID) {
  const meta = msgRoles.get(p.part.messageID);  // 查这个消息是谁发的
  if (!meta) return null;
  const text = p.part.text || "";
  if (meta.role === "user" && text) {
    return makePayload("UserPromptSubmit", meta.sessionID,
      sessionCwd.get(meta.sessionID), { prompt: text });
  }
  if (meta.role === "assistant" && text) {
    getSession(meta.sessionID).lastAssistantText = text;  // 只记不发
  }
  return null;
}
mapEvent — message.part.updated (tool) EVENT open-island.js
触发：模型调用了工具，工具开始执行或执行完毕。
做的事情：根据 status 判断是"开始用"还是"用完了"： pending/running → PreToolUse（告知桌面 App"要执行某某工具了"）；completed/error → PostToolUse。 
if (t === "message.part.updated" && p.part?.type === "tool" && p.part?.sessionID) {
  const st = p.part.state?.status;    // "pending" | "running" | "completed" | "error"
  const toolName = capitalize(p.part.tool);  // 工具名首字母大写

  if (st === "running" || st === "pending") {
    return makePayload("PreToolUse", p.part.sessionID, cwd, {
      tool_name: toolName,
      tool_input: p.part.state?.input       // 工具的参数
    });
  }
  if (st === "completed" || st === "error") {
    return makePayload("PostToolUse", p.part.sessionID, cwd, {
      tool_name: toolName
    });
  }
}
mapEvent — permission.asked EVENT open-island.js
触发：模型请求执行一个需要用户批准的操作（如运行 bash 命令、修改文件）。
做的事情：发 PermissionRequest 到桌面 App，显示对话框。
注意 _opencode_request_id — 用户批准后，插件用这个 ID 调用 HTTP API 告诉 opencode 结果。 
if (t === "permission.asked" && p.id && p.sessionID) {
  const toolName = capitalize(p.permission);  // "bash" / "edit" / "write"
  const patterns = p.patterns || [];

  const toolInput = { patterns, metadata: p.metadata };
  if (p.permission === "bash") {
    toolInput.command = patterns.join(" && ");  // 多条命令合并
  }
  if (p.permission === "edit" || p.permission === "write") {
    toolInput.file_path = patterns[0];          // 要修改的路径
  }

  return makePayload("PermissionRequest", p.sessionID, cwd, {
    tool_name: toolName,
    tool_input: toolInput,
    _opencode_request_id: p.id,  // 存起来，之后回应用
  });
}
mapEvent — session.status → idle EVENT open-island.js
触发：Agent 完成了所有工作，进入空闲状态。
做的事情：发 Stop 事件，附带助手最后一条消息。桌面 App 用这个更新 UI 状态。 
if (t === "session.status" && p.sessionID) {
  if (p.status?.type === "idle") {
    const s = getSession(p.sessionID);
    return makePayload("Stop", p.sessionID, sessionCwd.get(p.sessionID), {
      last_assistant_message: s.lastAssistantText  // 助手的最后一句
    });
  }
}

3. 工具函数

插件里反复用到的片段。
makePayload UTIL open-island.js
功能：统一构造发给桌面 App 的消息体。
所有事件最终都包装成 processOpenCodeHook 类型，openCodeHook 里带事件名、session_id、cwd。 
function makePayload(hookEventName, sessionID, cwd, extra = {}) {
  return {
    type: "processOpenCodeHook",
    openCodeHook: {
      hook_event_name: hookEventName,  // "SessionStart" / "UserPromptSubmit" / ...
      session_id: "opencode-" + sessionID,  // 统一加前缀
      cwd: cwd || ".",
      ...terminalFields(),  // 自动注入终端环境字段
      ...extra,             // 事件特定的额外字段
    },
  };
}
terminalFields + detectTty UTIL open-island.js
功能：检测用户正在使用的终端类型和 TTY 设备。
方法：从进程树向上遍历 8 层，找第一个不是 "?" 的 TTY。
支持：iTerm2、cmux、Zellij、Ghostty、Apple Terminal。 
// 检测 TTY 设备（从进程树向上走）
let detectedTty = null;
try {
  let walkPid = process.pid;
  for (let i = 0; i < 8; i++) {     // 最多往上翻 8 层
    const info = execSync("ps -o tty=,ppid= -p " + walkPid).toString().trim();
    const parts = info.split(/\s+/);
    const tty = parts[0], ppid = parseInt(parts[1]);
    if (tty && tty !== "??" && tty !== "?") {
      detectedTty = "/dev/" + tty; break;
    }
    if (!ppid || ppid <= 1) break;
    walkPid = ppid;  // 往父进程走
  }
} catch {}

// 根据环境变量判断终端种类
function terminalFields() {
  const env = process.env;
  if (env.ITERM_SESSION_ID)      return { terminal_app: "iTerm",     terminal_session_id: env.ITERM_SESSION_ID };
  if (env.GHOSTTY_RESOURCES_DIR) return { terminal_app: "Ghostty" };
  if (env.TERM_PROGRAM === "Apple_Terminal") return { terminal_app: "Terminal" };
  if (env.TERM_PROGRAM)           return { terminal_app: env.TERM_PROGRAM };
  return {};
}
normalizeQuestion UTIL open-island.js
功能：标准化 opencode 的"问题"格式。opencode 可能发各种格式的问题（字段名不统一），这个函数统一成 { question, header, options }。 
function normalizeQuestion(question, index) {
  if (!question || typeof question !== "object") return null;

  // 三种可能的字段名
  const questionText = question.question
                     || question.title
                     || question.prompt;
  if (!questionText) return null;

  // 选项也标准化：label / description / allowsFreeform
  const options = question.options
    .map(normalizeQuestionOption)
    .filter(Boolean);

  return {
    question: String(questionText),
    header: question.header || question.label || "Question " + (index + 1),
    options,
  };
}

function normalizeQuestionOption(option) {
  if (typeof option === "string") return { label: option };
  const label = option.label || option.text || option.value || option.name;
  if (!label) return null;
  return {
    label: String(label),
    description: option.description || option.hint || option.detail || "",
    allows_freeform: Boolean(option.allowsFreeform || option.allows_freeform),
  };
}

4. 插件骨架

这两个插件共同的最小结构。移除所有业务逻辑后，插件的"形状"只有这么一点。
export default FLOW 两个插件一致 
// 一个 opencode 插件 = 一个默认导出的 async 函数
// 返回两个 handler：event 和 shell.env
export default async function({ client, serverUrl, directory }) {
  // 初始化：连 socket、建缓存、记日志...

  return {
    "event": async function({ event }) {
      // opencode 每产生一个事件都会调这个函数
      // event.type: "session.created" | "message.part.updated" | "permission.asked" | ...
      const mapped = mapEvent(event);   // 转成自己的消息格式
      if (!mapped) return;

      if (需要用户交互) {
        sendAndWaitResponse(mapped).then(handleReply);  // 等用户答复
      } else {
        sendToSocket(mapped);  // 普通的通知，发完不管
      }
    },

    "shell.env": async function(input, output) {
      // opencode 每次创建子进程时调这个函数
      // 把终端环境变量注入子进程
      output.env.OPEN_ISLAND_ACTIVE = "1";
    },
  };
}

5. 总结：这些函数教会你什么

如果你要写自己的 opencode，你的框架需要：

  • 一个 event system：能在关键节点（session 开始、tool call、模型返回文字）发出事件
  • 一个 plugin 加载器：扫描目录、import JS 文件、调用 default export、传 { client, serverUrl, directory }
  • 一个 permission 系统：模型想跑 bash/改文件时，拦住、问用户、等回复
  • 一个 socket 通信：支持 fire-and-forget 和 wait-reply 两种模式

但这些都是 Month 3 的事。你现在只要能跑通 hello-llm 的 API 调用和 tool calling 循环就够——有了地基，上面这些东西只是选配。
